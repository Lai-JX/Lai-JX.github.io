                ordered = False
                break
        if ordered:
            return

        pos = self.partition(nums, l, r)
        self.quick_sort(nums, l, pos-1)
        self.quick_sort(nums, pos + 1, r)

        self.quick_sort(nums, 0, len(nums)-1)
        return nums

    Python 实现¶

    class Solution:
    def search(self, nums: List[int], target: int) -> int:
        left, right = 0, len(nums) - 1  # 定义target在左闭右闭的区间里，[left, right]

        while left <= right:
            middle = left + (right - left) // 2

            if nums[middle] > target:
                right = middle - 1  # target在左区间，所以[left, middle - 1]
            elif nums[middle] < target:
                left = middle + 1  # target在右区间，所以[middle + 1, right]
            else:
                return middle  # 数组中找到目标值，直接返回下标
        return -1  # 未找到目标值

    4. 寻找两个正序数组的中位数¶

    Hard 打开原题 本地补充实现

    快速复习¶

    把中位数转成寻找第 k 小；每轮排除一个数组中不可能包含答案的前 k/2 个元素。

    详细题解¶

    • 抓住寻找第k个元素的方法（比较或索引时采用的是闭区间）

    原笔记中的配图未随 Markdown 源文件保存。

    补充思路： 先确认是否有序、是否要求原地修改，再考虑二分、双指针、滑动窗口和前缀和。

    Python 实现¶

    class Solution:
    def findMedianSortedArrays(self, nums1, nums2):
        def kth(a, i, b, j, k):
            if i == len(a):
                return b[j + k - 1]
            if j == len(b):
                return a[i + k - 1]
            if k == 1:
                return min(a[i], b[j])
            half = k // 2
            ai = min(i + half, len(a)) - 1
            bj = min(j + half, len(b)) - 1
            if a[ai] <= b[bj]:
                return kth(a, ai + 1, b, j, k - (ai - i + 1))
            return kth(a, i, b, bj + 1, k - (bj - j + 1))

        total = len(nums1) + len(nums2)
        if total % 2:
            return kth(nums1, 0, nums2, 0, total // 2 + 1)
        left = kth(nums1, 0, nums2, 0, total // 2)
        right = kth(nums1, 0, nums2, 0, total // 2 + 1)
        return (left + right) / 2

    912. 排序数组¶

    Medium 打开原题 本地补充实现

    快速复习¶

    随机选择基准完成原地 partition；基准归位后只递归左右区间，平均时间 O(n log n)。

    详细题解¶

            class Solution:
            def partition(self, nums, l, r):
                i = randint(l, r)        # 随机选择基准
                nums[i], nums[r] = nums[r], nums[i]
                pivot = nums[r]
                i = l
                for j in range(l,r):
                    if nums[j] < pivot:
                        nums[i], nums[j] = nums[j], nums[i]
                        i += 1
                nums[i], nums[r] = nums[r], nums[i] # 最后一步一定要把基准归位
                return i
            def quick_sort(self, nums, l, r):
                if r-l < 1:
                    return
                ordered = True              # 优化：如果已经排序了就直接返回
                for i in range(l, r):
                    if nums[i] > nums[i + 1]:
                        ordered = False
                        break
                if ordered:
                    return

                pos = self.partition(nums, l, r)
                self.quick_sort(nums, l, pos-1)
                self.quick_sort(nums, pos + 1, r)

                return nums
                # return sorted(nums)

    Python 实现¶

    from random import randint

class Solution:
    def sortArray(self, nums):
        def quick_sort(left, right):
            if left >= right:
                return
            pivot_index = randint(left, right)
            nums[pivot_index], nums[right] = nums[right], nums[pivot_index]
            pivot = nums[right]
            store = left
            for i in range(left, right):
                if nums[i] < pivot:
                    nums[store], nums[i] = nums[i], nums[store]
                    store += 1
            nums[store], nums[right] = nums[right], nums[store]
            quick_sort(left, store - 1)
            quick_sort(store + 1, right)

        quick_sort(0, len(nums) - 1)
        return nums

    *1.3 有序数组的平方*

    这种题目关键在于数组中存在负数，其平方之后可能更大。可采用双指针，从数组两端向中间遍历，因为大的值出现在数组两端

    *1.4 长度最小的子数组*

    同样可以采用双指针的方法（这里有个更形象的名称叫“滑动窗口”）

    • 终止窗口位置（外层while）：扩大窗口，直至满足要求。（类似于快指针，用于遍历数组）
    • 起始窗口位置（内层while）：缩小窗口，直至无法满足长度要求

    1.5 螺旋矩阵II

    关键在于处理好循环（n//2次）、衔接（左闭右开）、求好offset（初始offset为1）、考虑奇数层的情况

    • 循环
      • 每次循环开始前把startx和starty赋值给i,j
      • 依据i和j进行赋值
      • 每次循环末尾更新startx和starty，以及offset

    Python 实现¶

    class Solution:
    def search(self, nums, target):
        left, right = 0, len(nums) - 1
        while left <= right:
            mid = (left + right) // 2
            if nums[mid] == target:
                return mid
            if nums[left] <= nums[mid]:
                if nums[left] <= target < nums[mid]:
                    right = mid - 1
                else:
                    left = mid + 1
            else:
                if nums[mid] < target <= nums[right]:
                    left = mid + 1
                else:
                    right = mid - 1
        return -1

    详细题解¶

    关键在于理解快慢指针

    • 快指针：用于遍历数组
    • 慢指针：指向更新 新数组下标的位置。
    • tip
      • 慢指针只有在赋值给新数组时（nums[fase_index]!=target），才自增
      • 而当nums[fase_index]==target ，只有快指针自增，表示跳过目标元素

    Python 实现¶

    class Solution:
    def removeElement(self, nums: List[int], val: int) -> int:
        i, l = 0, len(nums)
        while i < l:
            if nums[i] == val: # 找到等于目标值的节点
                for j in range(i+1, l): # 移除该元素，并将后面元素向前平移
                    nums[j - 1] = nums[j]
                l -= 1
                i -= 1
            i += 1
        return l

    详细题解¶

    同样可以采用双指针的方法（这里有个更形象的名称叫“滑动窗口”）

    • 终止窗口位置（外层while）：扩大窗口，直至满足要求。（类似于快指针，用于遍历数组）
    • 起始窗口位置（内层while）：缩小窗口，直至无法满足长度要求

    Python 实现¶

    class Solution:
    def minSubArrayLen(self, s: int, nums: List[int]) -> int:
        l = len(nums)
        min_len = float('inf')

        for i in range(l):
            cur_sum = 0
            for j in range(i, l):
                cur_sum += nums[j]
                if cur_sum >= s:
                    min_len = min(min_len, j - i + 1)
                    break

        return min_len if min_len != float('inf') else 0

    59.螺旋矩阵II¶

    Unknown 打开原题 代码随想录

    快速复习¶

    关键在于处理好循环（n//2次）、衔接（左闭右开）、求好offset（初始offset为1）、考虑奇数层的情况 - 循环 - 每次循环开始前把startx和starty赋值给i,j - 依据i和j进行赋值 - 每次循环末尾更新startx和starty，以及offset - 判断是否需要添加中心那一点

    详细题解¶

    关键在于处理好循环（n//2次）、衔接（左闭右开）、求好offset（初始offset为1）、考虑奇数层的情况

    • 循环
      • 每次循环开始前把startx和starty赋值给i,j
      • 依据i和j进行赋值
      • 每次循环末尾更新startx和starty，以及offset
    • 判断是否需要添加中心那一点

    Python 实现¶

    class Solution:
    def generateMatrix(self, n: int) -> List[List[int]]:
        nums = [[0] * n for _ in range(n)]
        startx, starty = 0, 0               # 起始点
        loop, mid = n // 2, n // 2          # 迭代次数、n为奇数时，矩阵的中心点
        count = 1                           # 计数

        for offset in range(1, loop + 1) :      # 每循环一层偏移量加1，偏移量从1开始
            for i in range(starty, n - offset) :    # 从左至右，左闭右开
                nums[startx][i] = count
                count += 1
            for i in range(startx, n - offset) :    # 从上至下
                nums[i][n - offset] = count
                count += 1
            for i in range(n - offset, starty, -1) : # 从右至左
                nums[n - offset][i] = count
                count += 1
            for i in range(n - offset, startx, -1) : # 从下至上
                nums[i][starty] = count
                count += 1                
            startx += 1         # 更新起始点
            starty += 1

        if n % 2 != 0 :         # n为奇数时，填充中心点
            nums[mid][mid] = count 
        return nums

    58. 区间和¶

    Unknown 打开原题 代码随想录

    快速复习¶

    其实利用前缀和就能解决；如果，我们想统计，在vec数组上 下标 2 到下标 5 之间的累加和，那是不是就用 p[5] - p[1] 就可以了

    详细题解¶

    其实利用前缀和就能解决。如果，我们想统计，在vec数组上 下标 2 到下标 5 之间的累加和，那是不是就用 p[5] - p[1] 就可以了。

    • 高难版：

    Python 实现¶

    import sys
input = sys.stdin.read

def main():
    data = input().split()
    index = 0
    n = int(data[index])
    index += 1
    vec = []
    for i in range(n):
        vec.append(int(data[index + i]))
    index += n

    p = [0] * n
    presum = 0
    for i in range(n):
        presum += vec[i]
        p[i] = presum

    results = []
    while index < len(data):
        a = int(data[index])
        b = int(data[index + 1])
        index += 2

        if a == 0:
            sum_value = p[b]
        else:
            sum_value = p[b] - p[a - 1]

        results.append(sum_value)

    for result in results:
        print(result)

if __name__ == "__main__":
    main()

    小苯的子串删除¶

    Unknown 打开原题 本地补充实现

    快速复习¶

    一个整数能被 3 整除等价于数位和能被 3 整除；把删除子串转成前缀和模 3 的配对计数。

    详细题解¶

    1.7 排序

    • 冒泡排序：一直向后交换，一轮下来就会有一个最大的数交换到后面所属位置
    • 选择排序：每一轮都选择最小的，然后放到所属位置
    • 插入排序：每一轮都从第i个位置向前交换，直至比前方的数大

    Python 实现¶

    # ACM 模式：统计删除一个连续子串后，剩余数字能被 3 整除的方案数
def solve():
    s = input().strip()
    prefix = [0]
    for ch in s:
        prefix.append((prefix[-1] + int(ch)) % 3)

    target = prefix[-1]
    count = [0, 0, 0]
    answer = 0
    for right in range(1, len(prefix)):
        count[prefix[right - 1]] += 1
        needed = (prefix[right] - target) % 3
        answer += count[needed]
    print(answer)

if __name__ == "__main__":
    solve()
