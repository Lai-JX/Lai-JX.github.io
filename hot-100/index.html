    from collections import defaultdict

class Solution:
    def groupAnagrams(self, strs):
        groups = defaultdict(list)
        for word in strs:
            count = [0] * 26
            for ch in word:
                count[ord(ch) - ord('a')] += 1
            groups[tuple(count)].append(word)
        return list(groups.values())

    128. 最长连续序列¶

    Medium 打开原题 Hot 100 补充

    快速复习¶

    把所有数字放入集合；只从没有前驱 x - 1 的数字开始向右扩展，每个数字最多访问一次。

    详细题解¶

    若 x - 1 不在集合中，x 一定是一段连续序列的起点。不断检查 x + 1、x + 2...，更新最长长度。跳过非起点可以避免对同一段序列重复扫描，平均时间复杂度为 O(n)。

    Python 实现¶

    class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        values = set(nums)
        answer = 0
        for value in values:
            if value - 1 in values:
                continue
            current = value
            while current in values:
                current += 1
            answer = max(answer, current - value)
        return answer

    283. 移动零¶

    Easy 打开原题 Hot 100 补充

    快速复习¶

    用慢指针维护下一个非零元素应写入的位置，先压缩非零元素，再把尾部补零。

    详细题解¶

    快指针顺序读取数组；遇到非零值就写到 slow 并递增。遍历结束后，[0, slow) 已按原相对顺序保存全部非零元素，将剩余位置写成零即可。

    Python 实现¶

    class Solution:
    def moveZeroes(self, nums: List[int]) -> None:
        slow = 0
        for value in nums:
            if value != 0:
                nums[slow] = value
                slow += 1
        while slow < len(nums):
            nums[slow] = 0
            slow += 1

    11. 盛最多水的容器¶

    Medium 打开原题 Hot 100 补充

    快速复习¶

    左右指针初始时指向首尾，计算面积，每次移动较短的一侧（保留短边不动不可能得到更大面积）。

    详细题解¶

    面积由宽度和两侧较短高度决定。指针向内移动后宽度必然减小，只有替换较短边才可能提高有效高度，因此每轮移动短边并更新最大值。

    Python 实现¶

    class Solution:
    def maxArea(self, height: List[int]) -> int:
        left, right = 0, len(height) - 1
        answer = 0
        while left < right:
            answer = max(answer, (right - left) * min(height[left], height[right]))
            if height[left] <= height[right]:
                left += 1
            else:
                right -= 1
        return answer

        • 如果三数之和太小，left++
        • 如果三数之和等于0，record，然后去重
    • 关键点：去重
      • 由于要求不重复，所以需要进行去重
      • 对于i，需要对nums[i]与nums[i-1]进行比较，而不能用nums[i]与nums[i+1]进行比较，因为后者还没遍历过，可能导致误删！同时需确保i-1有实际意义
      • 对于left，一直加1直至不相等
      • 对于right，一直减1直至不相等
      • 注意：得在去重之后再进行left和right更新。因为去重是和已遍历的数据比较

    Python 实现¶

    class Solution:
    def threeSum(self, nums: list[int]) -> list[list[int]]:
        nums = sorted(nums)
        ans = []
        n = len(nums)
        i = 0
        for i in range(n):
            if nums[i] > 0:
                break
            if i > 0 and nums[i-1] == nums[i]:      # 对i去重
                continue
            left = i + 1
            right = n - 1
            while left < right:
                tmp_sum = nums[i] + nums[left] + nums[right]
                if tmp_sum > 0:
                    right -= 1
                elif tmp_sum < 0:
                    left += 1
                else:
                    while left < right and nums[left] == nums[left+1]:      # 对left去重
                        left += 1
                    while left < right and nums[right] == nums[right-1]:    # 对right去重
                        right -= 1
                    ans.append([nums[i], nums[left], nums[right]])
                    left += 1   # 更新left
                    right -= 1  # 更新right
        return ans

    3. 无重复字符的最长子串¶

    Medium 打开原题 Hot 100 补充 滑动窗口

    快速复习¶

    滑动窗口记录字符最近出现位置；重复时让左边界跳到上次位置之后，但不能后退。

    详细题解¶

    右指针逐字符扩展窗口。若当前字符在窗口内出现过，将 left 更新为 max(left, last[ch] + 1)；随后记录最新位置并计算窗口长度。

    Python 实现¶

    class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        last = {}
        left = 0
        answer = 0
        for right, ch in enumerate(s):
            if ch in last:
                left = max(left, last[ch] + 1)
            last[ch] = right
            answer = max(answer, right - left + 1)
        return answer

    560. 和为 K 的子数组¶

    Medium 打开原题 Hot 100 补充

    快速复习¶

    前缀和 prefix 配合计数表；当前前缀需要寻找之前出现过多少次 prefix - k。

    详细题解¶

    若两个前缀和之差为 k，它们之间的子数组和就是 k。扫描时先累加 count[prefix-k]，再记录当前前缀；初始 {0: 1} 用于统计从下标 0 开始的子数组。

    Python 实现¶

    class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        count = {0: 1}
        prefix = 0
        answer = 0
        for value in nums:
            prefix += value
            answer += count.get(prefix - k, 0)
            count[prefix] = count.get(prefix, 0) + 1
        return answer

    239. 滑动窗口最大值¶

    返回原专题题解 · 栈与队列

    Hard 打开原题 代码随想录 滑动窗口

    快速复习¶

    维护值单调递减的队列；队首过期时若为最大则弹出，队尾不大于当前值时弹出。

    详细题解¶

    可使用单调队列（队列内单调递增或单调递减），以保证随时可以获取最大值

    1. push(value)：如果push的元素value大于入口元素的数值，那么就将队列入口的元素弹出，直到push元素的数值小于等于队列入口元素的数值为止
    2. pop(value)：如果窗口移除的元素value等于单调队列的出口元素，那么队列弹出元素，否则不用任何操作

    Python 实现¶

    from collections import deque
class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        queue = deque()
        ans = []
        for i in range(len(nums)):
            # push
            while queue and queue[-1] < nums[i]:
                queue.pop()
            queue.append(nums[i])

            # pop
            if i >= k and nums[i-k] == queue[0]:
                queue.popleft()

            # 更新结果
            if i >= k-1:
                ans.append(queue[0])
        return ans

    76. 最小覆盖子串¶

    Hard 打开原题 Hot 100 补充 双指针

    快速复习¶

    右指针扩展直到覆盖全部需求，左指针再尽量收缩；用 missing 记录仍缺少的字符总数。

    详细题解¶

    计数表need保存窗口还需要的字符数量。加入一个仍被需要的字符时 missing 减一；当 missing == 0 时持续移动左边界并更新最短答案，直到移走了必需字符。

    之所以只在 need[ch] > 0 时执行 missing -= 1，是因为这表示加入 ch 前窗口确实还缺少该字符；若 need[ch] <= 0，当前字符只是多余字符，不能减少缺失总数。

    移出左端字符时，先执行 need[outgoing] += 1；若此时 need[outgoing] > 0，说明移出后窗口重新缺少一个该字符，因此需要执行 missing += 1。若计数仍不大于零，则移出的只是多余字符，窗口依然满足需求。

    Python 实现¶

    from collections import Counter

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        need = Counter(t)
        missing = len(t)
        left = 0
        best_left, best_len = 0, float("inf")
        for right, ch in enumerate(s):
            if need[ch] > 0:
                missing -= 1
            need[ch] -= 1
            while missing == 0:
                if right - left + 1 < best_len:
                    best_left, best_len = left, right - left + 1
                outgoing = s[left]
                need[outgoing] += 1
                if need[outgoing] > 0:
                    missing += 1
                left += 1
        return "" if best_len == float("inf") else s[best_left:best_left + best_len]

        return res

    56. 合并区间¶

    返回原专题题解 · 贪心算法

    Medium 打开原题 代码随想录 贪心

    快速复习¶

    每个区间先按照左边界进行排序，再判断最后一个已合并区间的右边界是否与新区间的左边界相接。若相接则右边界更新为二者右边界的最大值

    详细题解¶

    和我们刚刚讲过的452. 用最少数量的箭引爆气球 (opens new window)和 435. 无重叠区间 (opens new window)都是一个套路。

    Python 实现¶

    class Solution:
    def merge(self, intervals):
        result = []
        if len(intervals) == 0:
            return result  # 区间集合为空直接返回

        intervals.sort(key=lambda x: x[0])  # 按照区间的左边界进行排序

        result.append(intervals[0])  # 第一个区间可以直接放入结果集中

        for i in range(1, len(intervals)):
            if result[-1][1] >= intervals[i][0]:  # 发现重叠区间
                # 合并区间，只需要更新结果集最后一个区间的右边界，因为根据排序，左边界已经是最小的
                result[-1][1] = max(result[-1][1], intervals[i][1])
            else:
                result.append(intervals[i])  # 区间不重叠

        return result

            prefix *= value

    41. 缺失的第一个正数¶

    返回原专题题解 · Hot 100

    Hard 打开原题 本地补充实现

    快速复习¶

    答案一定在 [1, n + 1] 中，可以复用原数组完成标记：

    • 原地置换：把值 x 放到下标 x - 1，最后找第一个 nums[i] != i + 1 的位置。
    • 负号标记：先排除非正数，再用 nums[x - 1] 的正负表示 x 是否出现。

    详细题解¶

    设数组长度为 n。如果 1 到 n 中有数没有出现，答案就是其中最小的一个；如果它们全部出现，答案只能是 n + 1。因此，负数、零以及大于 n 的数都不会影响答案。

    方法一：原地置换¶

    把数组本身当作一个哈希表：数字 x 应该放在下标 x - 1。遍历每个位置，只要当前数字属于 [1, n] 且尚未位于正确位置，就不断交换。

    交换条件中的 nums[nums[i] - 1] != nums[i] 用于处理重复数字，否则两个相同的数可能造成死循环。置换完成后，第一个不满足 nums[i] == i + 1 的位置对应的就是答案。

    虽然代码中有嵌套的 while，但每次交换都会让至少一个有效数字回到正确位置，总交换次数不超过 n，所以时间复杂度仍为 \(O(n)\)。

    方法二：负号标记¶

    先把所有非正数替换为 n + 1，避免它们干扰正负号。然后遍历数组：若当前数字的绝对值 x 在 [1, n] 中，就把 nums[x - 1] 变成负数，表示 x 出现过。

    读取时必须使用 abs(nums[i])，因为该位置可能已经被前面的数字改成负数。最后，第一个仍为正数的位置 i 表示 i + 1 没有出现；若全部为负数，则返回 n + 1。

    两种方法的时间复杂度都是 \(O(n)\)，额外空间复杂度都是 \(O(1)\)，并且都会修改原数组。

    Python 实现¶

    方法一：原地置换¶
    class Solution:
    def firstMissingPositive(self, nums: List[int]) -> int:
        n = len(nums)

        for i in range(n):
            while 1 <= nums[i] <= n and nums[nums[i] - 1] != nums[i]:
                target = nums[i] - 1
                nums[i], nums[target] = nums[target], nums[i]

        for i, value in enumerate(nums):
            if value != i + 1:
                return i + 1

        return n + 1
    方法二：负号标记¶
    class Solution:
    def firstMissingPositive(self, nums: List[int]) -> int:
        n = len(nums)

        # 非正数不可能成为答案，用范围外的正数替换。
        for i in range(n):
            if nums[i] <= 0:
                nums[i] = n + 1

        # nums[x - 1] < 0 表示数字 x 出现过。
        for i in range(n):
            x = abs(nums[i])
            if x <= n:
                nums[x - 1] = -abs(nums[x - 1])

        for i, value in enumerate(nums):
            if value > 0:
                return i + 1

        return n + 1

    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
        top, bottom = 0, len(matrix) - 1
        left, right = 0, len(matrix[0]) - 1
        answer = []
        while top <= bottom and left <= right:
            for col in range(left, right + 1):
                answer.append(matrix[top][col])
            top += 1
            for row in range(top, bottom + 1):
                answer.append(matrix[row][right])
            right -= 1
            if top <= bottom:
                for col in range(right, left - 1, -1):
                    answer.append(matrix[bottom][col])
                bottom -= 1
            if left <= right:
                for row in range(bottom, top - 1, -1):
                    answer.append(matrix[row][left])
                left += 1
        return answer

    • 采用双指针
      • 快指针先走n步，接着快慢指针一起走。当快指针到达最后一个节点时，慢指针指向要删除的节点

    # Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution:
    def removeNthFromEnd(self, head: ListNode, n: int) -> ListNode:
        # 创建一个虚拟节点，并将其下一个指针设置为链表的头部
        dummy_head = ListNode(0, head)

        # 创建两个指针，慢指针和快指针，并将它们初始化为虚拟节点
        slow = fast = dummy_head

        # 快指针比慢指针快 n+1 步
        for i in range(n+1):
            fast = fast.next

        # 移动两个指针，直到快速指针到达链表的末尾
        while fast:
            slow = slow.next
            fast = fast.next

        # 通过更新第 (n-1) 个节点的 next 指针删除第 n 个节点
        slow.next = slow.next.next

        return dummy_head.next

        self.size += 1

        node.next.prev = node.prev
        self.addToHead(node)

        self.head.next.prev = node
        self.head.next = node

    class Solution:
    def __init__(self):
        self.maxSum = float("-inf")

    def maxPathSum(self, root: TreeNode) -> int:
        def maxGain(node):
            if not node:
                return 0

            # 递归计算左右子节点的最大贡献值
            # 只有在最大贡献值大于 0 时，才会选取对应子节点
            leftGain = max(maxGain(node.left), 0)
            rightGain = max(maxGain(node.right), 0)

            # 节点的最大路径和取决于该节点的值与该节点的左右子节点的最大贡献值
            priceNewpath = node.val + leftGain + rightGain

            # 更新答案
            self.maxSum = max(self.maxSum, priceNewpath)

            # 返回节点的最大贡献值
            return node.val + max(leftGain, rightGain)

        maxGain(root)
        return self.maxSum

class Solution:
    def orangesRotting(self, grid: List[List[int]]) -> int:
        rows, cols = len(grid), len(grid[0])
        queue = deque()
        fresh = 0
        for row in range(rows):
            for col in range(cols):
                if grid[row][col] == 2:
                    queue.append((row, col))
                elif grid[row][col] == 1:
                    fresh += 1
        minutes = 0
        while queue and fresh:
            for _ in range(len(queue)):
                row, col = queue.popleft()
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = row + dr, col + dc
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 1:
                        grid[nr][nc] = 2
                        fresh -= 1
                        queue.append((nr, nc))
            minutes += 1
        return minutes if fresh == 0 else -1

        result = []  # 存储最终结果的二维字符串数组

        chessboard = ['.' * n for _ in range(n)]  # 初始化棋盘
        self.backtracking(n, 0, chessboard, result)  # 回溯求解

    def backtracking(self, n: int, row: int, chessboard: List[str], result: List[List[str]]) -> None:
        if row == n:
            result.append(chessboard[:])  # 棋盘填满，将当前解加入结果集
            return

        for col in range(n):
            if self.isValid(row, col, chessboard):
                chessboard[row] = chessboard[row][:col] + 'Q' + chessboard[row][col+1:]  # 放置皇后
                self.backtracking(n, row + 1, chessboard, result)  # 递归到下一行
                chessboard[row] = chessboard[row][:col] + '.' + chessboard[row][col+1:]  # 回溯，撤销当前位置的皇后

        for i in range(row):
            if chessboard[i][col] == 'Q':
                return False  # 当前列已经存在皇后，不合法

        # 检查 45 度角是否有皇后
        i, j = row - 1, col - 1
        while i >= 0 and j >= 0:
            if chessboard[i][j] == 'Q':
                return False  # 左上方向已经存在皇后，不合法
            i -= 1
            j -= 1

            if chessboard[i][j] == 'Q':
                return False  # 右上方向已经存在皇后，不合法

            j += 1

            left, right = 0, len(nums)
            while left < right:
                middle = (left + right) // 2
                if nums[middle] < value:
                    left = middle + 1
                else:
                    right = middle
            return left

        if first == len(nums) or nums[first] != target:
            return [-1, -1]

    class Solution:
    def longestCommonSubsequence(self, text1: str, text2: str) -> int:
        # 创建一个二维数组 dp，用于存储最长公共子序列的长度
        dp = [[0] * (len(text2) + 1) for _ in range(len(text1) + 1)]

        # 遍历 text1 和 text2，填充 dp 数组
        for i in range(1, len(text1) + 1):
            for j in range(1, len(text2) + 1):
                if text1[i - 1] == text2[j - 1]:
                    # 如果 text1[i-1] 和 text2[j-1] 相等，则当前位置的最长公共子序列长度为左上角位置的值加一
                    dp[i][j] = dp[i - 1][j - 1] + 1
                else:
                    # 如果 text1[i-1] 和 text2[j-1] 不相等，则当前位置的最长公共子序列长度为上方或左方的较大值
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])

        # 返回最长公共子序列的长度
        return dp[len(text1)][len(text2)]

    class Solution:
    def sortColors(self, nums: List[int]) -> None:
        left = current = 0
        right = len(nums) - 1
        while current <= right:
            if nums[current] == 0:
                nums[left], nums[current] = nums[current], nums[left]
                left += 1
                current += 1
            elif nums[current] == 2:
                nums[current], nums[right] = nums[right], nums[current]
                right -= 1
            else:
                current += 1

    def nextPermutation(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        length = len(nums)
        for i in range(length - 2, -1, -1): # 从倒数第二个开始
            if nums[i]>=nums[i+1]: continue # 剪枝去重

                    self.reverse(nums, i + 1, length - 1)

    def reverse(self, nums: List[int], left: int, right: int) -> None:
        while left < right:
            nums[left], nums[right] = nums[right], nums[left]
            left += 1
            right -= 1

        class Solution:
        def findDuplicate(self, nums: List[int]) -> int:
            n = len(nums)
            ans = 0
            for i in range(n):
                idx = abs(nums[i])-1
                if nums[idx] < 0:           # 第二次找到该位置
                    ans = idx
                    break
                nums[idx] = -1*nums[idx]    # 标记出现一次

            # 恢复原数组
            count = 0
            for i in range(n):
                idx = abs(nums[i])-1
                if ans == idx:
                    count += 1
                    if count == 2:
                        break
                nums[idx] = -1*nums[idx]    # 恢复

            return ans+1
